<script>
  // simple smooth scroll for anchor links and a tiny interactive console greeting
  document.querySelectorAll('a[href^="#"]').forEach(anchor => {
    anchor.addEventListener('click', function(e) {
      const targetId = this.getAttribute('href');
      if(targetId === "#" || targetId === "") return;
      const targetElement = document.querySelector(targetId);
      if(targetElement) {
        e.preventDefault();
        targetElement.scrollIntoView({ behavior: 'smooth', block: 'start' });
      }
    });
  });

  // demo product awareness: simple alert for shop button but not intrusive
  const shopButtons = document.querySelectorAll('.btn-secondary, .btn-primary');
  shopButtons.forEach(btn => {
    btn.addEventListener('click', (e) => {
      if(btn.innerText.includes('View full store') || btn.innerText.includes('Explore web builds')) {
        e.preventDefault();
        alert("✨ Welcome to Star Shipping demo! This is a marketing concept site. Real store & developer packages available upon contact.");
      } else if(btn.innerText.includes('Contact sales')) {
        e.preventDefault();
        alert("📧 Get in touch: hello@starshipping.dev — we’ll send you full catalog & special dev pricing.");
      } else if(btn.innerText.includes('Chat now')) {
        e.preventDefault();
        alert("💬 Live chat (demo): Our web dev team will reply within minutes. Real store builds start from $399.");
      }
    });
  });
</script>
